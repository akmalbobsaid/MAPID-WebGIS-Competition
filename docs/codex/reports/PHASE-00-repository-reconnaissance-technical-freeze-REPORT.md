### Python/PyPI compatibility

```powershell
$spatialPackages = 'geopandas','shapely','pyproj','networkx'
$results = foreach ($package in $spatialPackages) {
  $metadata = Invoke-RestMethod "https://pypi.org/pypi/$package/json"
  [pscustomobject]@{
    package = $package
    version = $metadata.info.version
    requires_python = $metadata.info.requires_python
    distribution_count = @($metadata.urls).Count
    cp313_win_amd64_wheels = @($metadata.urls | Where-Object { $_.filename -match 'cp313.*win_amd64' }).Count
    universal_wheels = @($metadata.urls | Where-Object { $_.filename -match 'py3-none-any' }).Count
  }
}
$results | Format-Table -AutoSize
$pythonCandidates = '3.13','3.12'

}
```
